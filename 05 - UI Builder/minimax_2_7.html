        function generateElementHTML(element) {
            const p = element.properties;

            switch (element.type) {
                case 'button':
                    return ``;
                case 'input':
                    return ``;
                case 'textarea':
                    return ``;
                case 'checkbox':
                    return `
✓
${p.text || 'Checkbox'}
`;
                case 'dropdown':
                    return `
${p.text || 'Dropdown'} ▾
`;
                case 'slider':
                    return `
${p.value || 50}%
`;
                case 'stepper':
                    return `
${p.value || 0}
`;
                case 'progress':
                    return `
`;
                case 'rectangle':
                    return ``;
                case 'circle':
                    const rx = Math.min(element.width, element.height) / 2;
                    return ``;
                case 'triangle':
                    return ``;
                case 'line':
                    return ``;
                case 'heading':
                    return `
${p.text || 'Heading'}
`;
                case 'paragraph':
                    return `
${p.text || 'Paragraph text...'}
`;
                default:
                    return '';
            }
        }

        function copyExportCode() {
            const code = document.getElementById('exportCode').textContent;
            navigator.clipboard.writeText(code).then(() => {
                alert('Code copied to clipboard!');
            });
        }

        function downloadExportCode() {
            const code = document.getElementById('exportCode').textContent;
            const blob = new Blob([code], { type: 'text/html' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'ui-export.html';
            a.click();
            URL.revokeObjectURL(url);
        }
